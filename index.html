/** * AKS Khan Healthcare Ltd. — Shared Google Sheets backend * for the combined Meeting Room Booking + Stationery Requisition portal (index.html). * * This script does TWO jobs: * 1. A simple key-value store (KV_Store sheet) backing both modules' shared data. * 2. Sending booking status emails (Approved / Rejected / Cancelled) to employees * when the admin acts on a Meeting Room booking. * * SETUP * 1. Create a new Google Sheet (sheets.google.com -> Blank). * 2. Extensions -> Apps Script. * 3. Delete any starter code, paste this whole file in, and save. * 4. Click Deploy -> New deployment. * - Select type: Web app * - Description: anything * - Execute as: Me <-- IMPORTANT: use the admin.ho@akskhanpharma.com Google account * here if you want emails to be sent FROM that address. * - Who has access: Anyone * 5. Click Deploy, authorize the permissions Google asks for (including "Send email * as you" — this is required for the approval emails to work). * 6. Copy the "https://script.google.com/macros/s/AKfycbwez_L-3B-h05n5_JNHLpUGZBAgC7YHJUgLEaFmQ5l8xdd5M1q9IpfyR72eoZi115XR/exec" it gives you. * 7. Open index.html, find window.SHEETS_API_URL near the very top of the file, * and paste your URL between the quotes. * * IMPORTANT — IF YOU EDIT THIS CODE LATER: * Saving the script is NOT enough. You must create a NEW deployment version: * Deploy -> Manage deployments -> (pencil/edit icon) -> Version: New version -> Deploy * Otherwise your live portal keeps talking to the OLD code. */ var SHEET_NAME = 'KV_Store'; function doGet(e) { var key = e.parameter.key; var sheet = getSheet_(); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(key)) { return jsonOutput_({ key: key, value: data[i][1] }); } } return jsonOutput_({ key: key, value: null }); } function doPost(e) { var body = JSON.parse(e.postData.contents); if (body.action === 'sendBookingEmail') { return handleSendBookingEmail_(body); } // Default: key-value store write (used by both portal modules) var key = body.key; var value = body.value; var sheet = getSheet_(); var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]) === String(key)) { sheet.getRange(i + 1, 2).setValue(value); sheet.getRange(i + 1, 3).setValue(new Date()); return jsonOutput_({ key: key, value: value, updated: true }); } } sheet.appendRow([key, value, new Date()]); return jsonOutput_({ key: key, value: value, created: true }); } function handleSendBookingEmail_(body) { try { var to = body.to; var status = body.status || 'Updated'; // Approved / Rejected / Cancelled var subject = 'Meeting Room Booking ' + status; var message = 'Dear ' + body.employeeName + ',\n\n' + 'Your meeting room booking request has been ' + status.toLowerCase() + '.\n\n' + 'Meeting Room: ' + body.roomName + '\n' + 'Date: ' + body.date + '\n' + 'Time: ' + body.startTime + ' \u2013 ' + body.endTime + '\n\n' + 'Please proceed accordingly.\n\n' + 'Thanks & Regards,\n' + 'Administration'; MailApp.sendEmail({ to: to, subject: subject, body: message, name: 'A K S Khan Healthcare Ltd. — Administration' }); return jsonOutput_({ ok: true }); } catch (err) { return jsonOutput_({ ok: false, error: String(err) }); } } function getSheet_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['key', 'value', 'updated_at']); sheet.setFrozenRows(1); } return sheet; } function jsonOutput_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
